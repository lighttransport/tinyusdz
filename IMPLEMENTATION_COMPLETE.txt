USAGE EXAMPLE
================================================================================

// Load USD file with variants
tinyusdz::Stage stage;
tinyusdz::LoadUSDFromFile("scene.usda", &stage, ...);

// Convert to RenderScene (with variants)
tinyusdz::tydra::RenderSceneConverter converter;
tinyusdz::tydra::RenderScene render_scene;
converter.ConvertToRenderScene(env, &render_scene);

// Query available variants
for (const auto& group : render_scene.variant_groups) {
    std::cout << "Prim: " << group.prim_path << "\n";
    for (const auto& vset : group.variant_sets) {
        std::cout << "  VariantSet: " << vset.name << "\n";
        for (const auto& opt : vset.options) {
            std::cout << "    Option: " << opt.name << "\n";
        }
    }
}

// Switch variants at runtime
render_scene.active_selections.push_back({
    "color", "red", "/Car"
});

// Apply variant changes to rendering...
